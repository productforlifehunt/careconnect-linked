(function() {
  const baseUrl = '/careconnected/wp-json/wp/v2';
  
  // Get current user ID from WordPress
  async function getCurrentUserId() {
    try {
      const res = await fetch('/careconnected/wp-json/wp/v2/users/me', {
        credentials: 'include'
      });
      if (!res.ok) return null;
      const user = await res.json();
      return user.id;
    } catch (e) {
      console.error('Failed to get current user:', e);
      return null;
    }
  }
  
  async function loadStats() {
    const userId = await getCurrentUserId();
    if (!userId) {
      console.log('User not logged in');
      return;
    }
    
    try {
      // Fetch bookings count (wc_booking CPT, author = current user)
      const bookingsRes = await fetch(`${baseUrl}/wc_booking?author=${userId}&status=confirmed,pending&per_page=1`, {
        credentials: 'include'
      });
      const bookingsTotal = bookingsRes.headers.get('X-WP-Total') || '0';
      document.getElementById('stat-bookings').textContent = bookingsTotal;
      
      // Fetch care groups count (care_group CPT, author = current user)
      const groupsRes = await fetch(`${baseUrl}/care_group?author=${userId}&per_page=1`, {
        credentials: 'include'
      });
      const groupsTotal = groupsRes.headers.get('X-WP-Total') || '0';
      document.getElementById('stat-groups').textContent = groupsTotal;
      
      // Fetch pending tasks count (care_task CPT, author = current user, status = pending)
      const tasksRes = await fetch(`${baseUrl}/care_task?author=${userId}&status=pending&per_page=1`, {
        credentials: 'include'
      });
      const tasksTotal = tasksRes.headers.get('X-WP-Total') || '0';
      document.getElementById('stat-tasks').textContent = tasksTotal;
      
      // Fetch unread messages count (chat_message CPT)
      const messagesRes = await fetch(`${baseUrl}/chat_message?per_page=1&meta_key=is_read&meta_value=0`, {
        credentials: 'include'
      });
      const messagesTotal = messagesRes.headers.get('X-WP-Total') || '0';
      document.getElementById('stat-messages').textContent = messagesTotal;
      
    } catch (e) {
      console.error('Failed to load stats:', e);
    }
  }
  
  // Load stats on page load
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', loadStats);
  } else {
    loadStats();
  }
})();
</script>

<script>
(function() {
  const baseUrl = '/careconnected/wp-json/wp/v2';
  
  async function getCurrentUserId() {
    try {
      const res = await fetch('/careconnected/wp-json/wp/v2/users/me', {
        credentials: 'include'
      });
      if (!res.ok) return null;
      const user = await res.json();
      return user.id;
    } catch (e) {
      return null;
    }
  }
  
  function formatDate(dateStr) {
    const date = new Date(dateStr);
    const options = { weekday: 'short', month: 'short', day: 'numeric' };
    return date.toLocaleDateString('en-US', options);
  }
  
  async function loadBookings() {
    const userId = await getCurrentUserId();
    if (!userId) {
      document.getElementById('bookings-list').innerHTML = '<div style="text-align: center; padding: 2rem; color: #9ca3af;"><div style="font-size: 2rem; margin-bottom: 0.5rem;">🔒</div><div>Please log in to view bookings</div></div>';
      return;
    }
    
    try {
      const res = await fetch(`${baseUrl}/wc_booking?author=${userId}&status=confirmed,pending&per_page=5&orderby=date&order=asc&_embed`, {
        credentials: 'include'
      });
      
      if (!res.ok) throw new Error('Failed to fetch bookings');
      
      const bookings = await res.json();
      
      if (bookings.length === 0) {
        document.getElementById('bookings-list').innerHTML = '<div style="text-align: center; padding: 2rem; color: #9ca3af;"><div style="font-size: 2rem; margin-bottom: 0.5rem;">📅</div><div>No upcoming bookings</div></div>';
        return;
      }
      
      const html = bookings.map(booking => {
        const title = booking.title?.rendered || 'Booking';
        const date = booking.meta?._booking_start || booking.date;
        const status = booking.status || 'pending';
        const statusColors = {
          'confirmed': '#22c55e',
          'pending': '#f59e0b',
          'completed': '#6b7280',
          'cancelled': '#ef4444'
        };
        const statusColor = statusColors[status] || '#6b7280';
        
        return `
          <div style="padding: 1rem; border-bottom: 1px solid #e5e7eb; display: flex; justify-content: space-between; align-items: center;">
            <div>
              <div style="font-weight: 500; color: #111827;">${title}</div>
              <div style="font-size: 0.875rem; color: #6b7280; margin-top: 0.25rem;">${formatDate(date)}</div>
            </div>
            <div style="background: ${statusColor}; color: white; padding: 0.25rem 0.75rem; border-radius: 9999px; font-size: 0.75rem; font-weight: 500;">
              ${status}
            </div>
          </div>
        `;
      }).join('');
      
      document.getElementById('bookings-list').innerHTML = html;
      
    } catch (e) {
      console.error('Failed to load bookings:', e);
      document.getElementById('bookings-list').innerHTML = '<div style="text-align: center; padding: 2rem; color: #ef4444;"><div>Failed to load bookings</div></div>';
    }
  }
  
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', loadBookings);
  } else {
    loadBookings();
  }
})();
</script>
